export function normalizeIngredient(raw: string): string
// Returns '' for empty input
// 10-step pipeline:
//   1. lowercase + trim
//   2. strip parentheticals e.g. "(about 2 cups)"
//   3. strip stray punctuation
//   4. collapse whitespace
//   5. check PROTECTED_COMPOUNDS (e.g. "ground beef" skips prefix stripping)
//   6. strip PREP_PREFIXES (diced, chopped, fresh, etc.) — loops for chained
//   7. strip PREP_SUFFIXES (divided, optional, garnish, etc.)
//   8. depluralize last word (berries→berry, onions→onion, tomatoes→tomato)
//   9. synonym lookup (cilantro→coriander, evoo→olive oil, etc.)
//  10. final trim

export function pickDisplayName(variants: string[]): string
// Returns '' for empty array
// Returns titleCase(variants[0]) for single-item array
// For multiple: picks most frequent lowercase variant (mode), breaks ties with shortest
// Returns title-cased result
```

export function formatQuantity(value: number): string
// value <= 0 → ''
// Integer → string (e.g. 3 → '3')
// Matches FRACTION_MAP within tolerance 0.05:
//   [0.125,'⅛'], [0.25,'¼'], [0.333,'⅓'], [0.375,'⅜'],
//   [0.5,'½'], [0.625,'⅝'], [0.667,'⅔'], [0.75,'¾'], [0.875,'⅞']
// whole + fraction: e.g. 1.5 → '1 ½', 2.25 → '2 ¼'
// No match: round to 1 decimal (e.g. 1.6 → '1.6')
// Edge: 1.333 → '1 ⅓', 0.667 → '⅔', 3.875 → '3 ⅞'

export function getScaleFactor(baseServings: number, desiredServings: number): number
// baseServings <= 0 → 1 (safe default)
// Otherwise: desiredServings / baseServings

export function scaleQuantity(rawQuantity: string, factor: number): string
// Non-numeric ('to taste', 'pinch', 'a handful') → returns rawQuantity unchanged
// Numeric string: parses, multiplies by factor, formats via formatQuantity
// '0' or negative-parsed → returns rawQuantity unchanged
// '2' × 1.5 → '3'

export function scaleIngredients(ingredients: Ingredient[], baseServings: number, desiredServings: number): Ingredient[]
// Returns original array if factor === 1 (identity optimization)
// Returns new array (does NOT mutate original)
// Each ingredient: quantity scaled, name and unit preserved

export function scaleCalories(totalCalories: number | null, baseServings: number, desiredServings: number): number | null
// null input → null
// baseServings <= 0 → returns totalCalories unchanged
// Scales proportionally and rounds: 500cal/4serv × 2 = 250

export function caloriesPerServing(totalCalories: number | null, servings: number): number | null
// null → null
// servings <= 0 → null
// 500/4 = 125, rounds to nearest integer
```
